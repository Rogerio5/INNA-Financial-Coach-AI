00429:         duracao_b - duracao_a

00433:         (
00434:             diferenca_absoluta

00439:         else 0.0
00440:     )
00441: 
00442:     return {
00443:         "diferenca_absoluta_ms": round(
00444:             diferenca_absoluta,
