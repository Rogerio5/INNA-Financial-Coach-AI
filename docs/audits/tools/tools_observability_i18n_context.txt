def calcular_taxa_sucesso(
    total_sucesso: int,
    total_executadas: int,
) -> float:
    if total_executadas <= 0:
        return 0.0

    return round(
        (total_sucesso / total_executadas) * 100,
        2,
    )



===== PREPARAÇÃO DO RANKING =====

def preparar_ranking_tools(
    df: pd.DataFrame | None,
) -> pd.DataFrame:
    if df is None or df.empty:
        return pd.DataFrame()

    base = df.copy()

    base["tool_label"] = (
        base["tool_label"]
        .fillna(base["tool_nome"])
        .fillna("Sem Tool")
    )

    ranking = (
        base.groupby("tool_label")
        .agg(
            Execuções=("id", "count"),
            Sucessos=("sucesso", "sum"),
            Falhas=(
                "sucesso",
                lambda valores: int(
                    (~valores.astype(bool)).sum()
                ),
            ),
            Média_ms=("duracao_ms", "mean"),
            Mínimo_ms=("duracao_ms", "min"),
            Máximo_ms=("duracao_ms", "max"),
        )
        .reset_index()
        .rename(
            columns={
                "tool_label": "Ferramenta",
            }
        )
    )

    ranking["Taxa de sucesso (%)"] = ranking.apply(
        lambda row: calcular_taxa_sucesso(
            int(row["Sucessos"]),
            int(row["Execuções"]),
        ),
        axis=1,
    )

    for coluna in [
        "Média_ms",
        "Mínimo_ms",
        "Máximo_ms",
    ]:
        ranking[coluna] = (
            ranking[coluna]
            .fillna(0)
            .round(3)
        )

    return ranking.sort_values(
        by=["Execuções", "Taxa de sucesso (%)"],
        ascending=[False, False],
    ).reset_index(drop=True)

def preparar_eventos_exibicao(
    df: pd.DataFrame,
) -> pd.DataFrame:
    if df.empty:
        return pd.DataFrame()

    tabela = df.copy()

    tabela["Ferramenta"] = (
        tabela["tool_label"]
        .fillna(tabela["tool_nome"])
        .fillna("Sem Tool")
    )

    tabela["Status"] = tabela["sucesso"].map({
        True: "Sucesso",
        False: "Falha",
    })

    tabela["Confiança"] = (
        tabela["confianca"]
        .fillna(0)
        .apply(
            lambda valor: f"{float(valor) * 100:.0f}%"
        )
    )

    tabela["Duração"] = (
        tabela["duracao_ms"]
        .fillna(0)
        .apply(
            lambda valor: f"{float(valor):.3f} ms"
        )
    )

    tabela["Data"] = pd.to_datetime(
        tabela["criado_em"],
        errors="coerce",
    ).dt.strftime("%d/%m/%Y %H:%M:%S")

    colunas = [
        "Data",
        "usuario_id",
        "pergunta",
        "Ferramenta",
        "intencao",
        "Confiança",
        "Status",
        "Duração",
        "origem",
        "mensagem",
        "erro",
    ]

    return tabela[
        [
            coluna
            for coluna in colunas
            if coluna in tabela.columns
        ]
    ].rename(
        columns={
            "usuario_id": "Usuário",
            "pergunta": "Pergunta",
            "intencao": "Intenção",
            "origem": "Origem",
            "mensagem": "Mensagem",
            "erro": "Erro",
        }
    )


def gerar_excel(
    df_eventos: pd.DataFrame,
    df_ranking: pd.DataFrame,
) -> bytes:
    output = BytesIO()

    with pd.ExcelWriter(
        output,
        engine="openpyxl",
    ) as writer:
        df_eventos.to_excel(
            writer,
            sheet_name="Eventos",
            index=False,
        )

        df_ranking.to_excel(
            writer,
            sheet_name="Ranking",
            index=False,
        )

    return output.getvalue()

def render_metricas_executivas(
    df: pd.DataFrame,
) -> dict:
    """
    Renderiza os KPIs executivos do Operations Center.
    """
    metricas = calcular_metricas_executivas(
        df
    )

    st.markdown("## Visão executiva")

    linha_1 = st.columns(4)

    linha_1[0].metric(
        "Eventos registrados",
        metricas["total_eventos"],
    )

    linha_1[1].metric(
        "Execuções hoje",
        metricas["execucoes_hoje"],
    )

    linha_1[2].metric(
        "Últimas 24 horas",
        metricas["execucoes_ultimas_24h"],
    )

    linha_1[3].metric(
        "Usuários ativos",
        metricas["usuarios_ativos"],
    )

    linha_2 = st.columns(4)

    linha_2[0].metric(
        "Taxa de sucesso",
        (
            f"{metricas[
                'taxa_sucesso_percentual'
            ]:.2f}%"
        ),
    )

    linha_2[1].metric(
        "Execuções com sucesso",
        metricas["total_sucesso"],
    )

    linha_2[2].metric(
        "Falhas",
        metricas["total_falhas"],
    )

    linha_2[3].metric(
        "Confiança média",
        (
            f"{metricas[
                'confianca_media_percentual'
            ]:.2f}%"
        ),
    )

    linha_3 = st.columns(4)

    linha_3[0].metric(
        "Duração média",
        (
            f"{metricas[
                'duracao_media_ms'
            ]:.3f} ms"
        ),
    )

    linha_3[1].metric(
        "Duração mínima",
        (
            f"{metricas[
                'duracao_minima_ms'
            ]:.3f} ms"
        ),
    )

    linha_3[2].metric(
        "Duração P95",
        (
            f"{metricas[
                'duracao_p95_ms'
            ]:.3f} ms"
        ),
        help=(
            "95% das execuções terminaram "
            "abaixo deste tempo."
        ),
    )

    linha_3[3].metric(
        "Duração máxima",
        (
            f"{metricas[
                'duracao_maxima_ms'
            ]:.3f} ms"
        ),
    )

    tool_top = metricas.get(
        "tool_mais_utilizada",
        {},
    )

    ultima_execucao = metricas.get(
        "ultima_execucao"
    )

    col_destaque_1, col_destaque_2 = st.columns(2)

    with col_destaque_1:
        st.markdown("### Tool mais utilizada")

        if tool_top.get("nome"):
            st.success(
                f"{tool_top['nome']} — "
                f"{tool_top['total']} execução(ões)"
            )
        else:
            st.info(
                "Ainda não há Tool predominante."
            )

    with col_destaque_2:
        st.markdown("### Última execução")

        if ultima_execucao:
            data_formatada = pd.to_datetime(
                ultima_execucao,
                errors="coerce",
            )

            if pd.notna(data_formatada):
                st.info(
                    data_formatada.strftime(
                        "%d/%m/%Y %H:%M:%S"
                    )
                )
            else:
                st.info(str(ultima_execucao))
        else:
            st.info(
                "Nenhuma execução registrada."
            )

    return metricas


def render_status_infraestrutura(
    resultado_status: dict[str, Any],
) -> None:
    st.markdown("## Estado da infraestrutura")

    status = resultado_status.get(
        "status",
        {},
    )

    total = resultado_status.get(
        "total_componentes",
        0,
    )

    operacionais = resultado_status.get(
        "total_operacionais",
        0,
    )

    st.progress(
        operacionais / total
        if total
        else 0
    )

    st.caption(
        f"{operacionais} de {total} componentes operacionais."
    )

    colunas = st.columns(3)

    for indice, item in enumerate(
        status.values()
    ):
        coluna = colunas[
            indice % len(colunas)
        ]

        simbolo = "🟢" if item["ok"] else "🔴"

        coluna.markdown(
            f"**{simbolo} {item['label']}**"
        )

        coluna.caption(
            item["message"]
        )
